cmake_minimum_required(VERSION 3.22.1)
project(talker_jni LANGUAGES C CXX)

set(AUDIOCPP_LIB "${CMAKE_CURRENT_SOURCE_DIR}/../jniLibs/arm64-v8a/libaudiocpp.so")
set(AUDIOCPP_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/include")
if(NOT EXISTS "${AUDIOCPP_LIB}" OR NOT EXISTS "${AUDIOCPP_INCLUDE}/audiocpp.h")
    message(FATAL_ERROR "Prebuilt Android audio.cpp C API library/header missing")
endif()

add_library(audiocpp SHARED IMPORTED)
set_target_properties(audiocpp PROPERTIES IMPORTED_LOCATION "${AUDIOCPP_LIB}")
add_library(talker_jni SHARED native-lib.cpp)
target_include_directories(talker_jni PRIVATE "${AUDIOCPP_INCLUDE}")
target_compile_features(talker_jni PRIVATE cxx_std_17)
target_link_libraries(talker_jni PRIVATE audiocpp log)
